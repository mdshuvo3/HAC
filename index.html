function sendWifiUpdate() { const ssid = document.getElementById('new-ssid').value.trim(); const pass = document.getElementById('new-pass').value.trim(); if (!ssid) return alert("SSID cannot be empty!"); if (confirm("Send new Wi-Fi credentials? ESP32 will attempt to reboot and connect.")) { client.publish(mqttCmdTopic, JSON.stringify({ action: "update_wifi", ssid: ssid, pass: pass })); toggleModal(false); } }
